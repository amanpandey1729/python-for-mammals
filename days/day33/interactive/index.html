def run_health_check(settings, scenario):
    collected = collect_scenario(scenario)

    cpu = parse_cpu_percent(collected["mpstat"])
    uptime = parse_uptime(collected["uptime"])
    memory = parse_memory(collected["free"])
    filesystems = parse_disk(collected["df"])
    processes = parse_processes(
        collected["ps"],
        settings["top_process_count"],
    )

    alerts = []

    if cpu >= settings["cpu_threshold"]:
        alerts.append(
            f"CPU utilisation is {cpu}%"
        )

    load_limit = (
        collected["cpu_count"]
        * settings["load_warning_per_cpu"]
    )

    if uptime["load_1m"] >= load_limit:
        alerts.append(
            f"One-minute load is {uptime['load_1m']}"
        )

    if memory["memory_percent"] >= settings["memory_threshold"]:
        alerts.append(
            f"Memory utilisation is {memory['memory_percent']}%"
        )

    for filesystem in filesystems:
        if filesystem["used_percent"] >= settings["disk_threshold"]:
            alerts.append(
                f"{filesystem['mount_point']} is "
                f"{filesystem['used_percent']}% used"
            )

    final_status = (
        "ATTENTION" if alerts else "HEALTHY"
    )

    return build_report(
        cpu,
        uptime,
        memory,
        filesystems,
        processes,
        alerts,
        final_status,
    )
